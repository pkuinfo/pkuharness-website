## Types and schemas

```ts
/** "HH:MM", 24-hour, local time. */
const timeOfDay = z.string().regex(/^([01]?\d|2[0-3]):[0-5]\d$/, 'time must be HH:MM (24h)');

const triggerSchema = z.discriminatedUnion('type', [
  z.object({ type: z.literal('at'), at: z.number() /* epoch ms */ }),
  z.object({
    type: z.literal('every'),
    unit: z.enum(['minute', 'hour', 'day', 'week']),
    interval: z.number().int().min(1).max(1000),
    /** day/week units: times of day; default = the time of `startAt`. */
    times: z.array(timeOfDay).min(1).max(24).optional(),
    /** week unit: 0 = Sunday … 6 = Saturday; default = weekday of `startAt`. */
    weekdays: z.array(z.number().int().min(0).max(6)).min(1).max(7).optional(),
    startAt: z.number(), // epoch ms; first possible occurrence
    /** Total occurrences in the series, then stop. */
    count: z.number().int().min(1).optional(),
    /** Last possible occurrence (inclusive), epoch ms. */
    until: z.number().optional(),
  }),
]);
type Trigger = z.infer<typeof triggerSchema>;

const conditionSchema = z.object({
  /** Dot path into the data source result, e.g. "balance". Empty = the whole result. */
  path: z.string().default(''),
  op: z.enum(['<', '<=', '>', '>=', '==', '!=', 'changed', 'always']),
  value: z.union([z.number(), z.string(), z.boolean()]).optional(),
});
type Condition = z.infer<typeof conditionSchema>;

const actionSchema = z.discriminatedUnion('type', [
  z.object({ type: z.literal('notify'), title: z.string().min(1), body: z.string().optional(), url: z.string().optional() }),
  z.object({
    type: z.literal('check'),
    tool: z.string(),                                      // `<plugin>.<tool>`, read-only
    args: z.record(z.string(), z.unknown()).optional(),
    condition: conditionSchema,
    title: z.string().min(1),                              // `{value}` is replaced by the checked value
    body: z.string().optional(),
    url: z.string().optional(),
  }),
  z.object({ type: z.literal('agent'), prompt: z.string().min(4).max(1000) }),
]);
type Action = z.infer<typeof actionSchema>;

type JobSource = 'user' | 'agent' | 'plugin';

type JobRun = {
  at: number;
  /** The occurrence it ran for (null for a manual run). */
  scheduledAt: number | null;
  ok: boolean;
  notified: boolean;
  value?: unknown;
  error?: string;
  ms: number;
  manual?: boolean;
  /** `agent` jobs: the reply (empty when it stayed silent). */
  output?: string;
  /** Not run: more than MAX_LATE_MS late. */
  missed?: boolean;
};

type Job = {
  id: string;
  title: string;
  trigger: Trigger;
  action: Action;
  enabled: boolean;
  source: JobSource;
  runs: number;
  lastRunAt: number | null;
  nextRunAt: number | null;
  lastValue?: unknown;
  lastError?: string;
  history?: JobRun[];
  failures?: number;
  alertedAt?: number | null;
  meta?: Record<string, unknown>;
  createdAt: number;
  updatedAt: number;
};

/** Jobs whose runs execute in JS (a check or an agent run) rather than as a pre-armed notification. */
const runsInJs: (action: Action) => boolean;
```

All of `timeOfDay`, `triggerSchema`, `conditionSchema` and `actionSchema` are exported.

## Recurrence engine

```ts
type Occurrence = { at: number; index: number };

function nextOccurrence(trigger: Trigger, after: number): Occurrence | null;           // first strictly after `after`
function occurrencesAfter(trigger: Trigger, after: number, limit: number, before?: number): Occurrence[];
function lastOccurrence(trigger: Trigger, t: number): Occurrence | null;               // latest at or before t
function runsBy(trigger: Trigger, t: number): number;                                  // occurrences by t
function totalOccurrences(trigger: Trigger): number | null;                            // null = open-ended
function describeTrigger(trigger: Trigger, now?: number, locale?: 'zh-Hans' | 'en'): string;
function parseTime(t: string): [number, number];                                       // "08:30" → [8, 30]
const formatTime: (h: number, m: number) => string;                                    // (8, 5) → "08:05"
```

A series is a sequence of periods (one minute/hour step, one day, or one Monday-based week), each
with a fixed list of slots (times × weekdays). Slots before `startAt` are skipped; the rest are
numbered 0, 1, 2… and `count` / `until` cut the series. DST-safe.

```ts
const every3Days16Times: Trigger = {
  type: 'every', unit: 'day', interval: 3, times: ['09:00'], count: 16,
  startAt: new Date(2026, 9, 20).getTime(),
};
occurrencesAfter(every3Days16Times, Date.now(), 3);   // next three
totalOccurrences(every3Days16Times);                  // 16
describeTrigger(every3Days16Times, Date.now(), 'en'); // e.g. "Every 3 days at 09:00 · 16 times"
```

## ScheduleInput (model- and form-friendly)

```ts
const scheduleInputSchema = z.discriminatedUnion('type', [
  z.object({ type: z.literal('once'), at: localDateTime }),
  z.object({
    type: z.literal('repeat'),
    unit: z.enum(['minute', 'hour', 'day', 'week']),
    interval: z.number().int().min(1),          // every N units
    times: z.array(timeOfDay).optional(),       // day/week
    weekdays: z.array(z.number().int().min(0).max(6)).optional(),  // week: 0 = Sunday
    start: localDateTime.optional(),            // default now
    count: z.number().int().min(1).optional(),
    until: localDateTime.optional(),            // inclusive
  }),
]);
// localDateTime: /^\d{4}-\d{2}-\d{2}[ T]\d{1,2}:\d{2}(:\d{2})?$/  ("YYYY-MM-DD HH:mm")
type ScheduleInput = z.infer<typeof scheduleInputSchema>;

function toTrigger(input: ScheduleInput, now?: number): Trigger;
function toScheduleInput(t: Trigger): ScheduleInput;
function parseLocal(s: string): number;      // throws on bad or non-existent dates
function formatLocal(ms: number): string;    // "2026-10-20 08:00 周二"
function summarize(trigger: Trigger, now?: number, n?: number): {
  schedule: string;       // describeTrigger
  next: string[];         // next n occurrences, formatLocal
  total: number | null;
  finished: boolean;
};
```

Use `scheduleInputSchema` as a tool input and `summarize` in the result, so the model can read back
the exact times to the user.

## Conditions

```ts
/** Read a dot path ("data.balance", "items.0.amount") from a tool result. */
function getPath(value: unknown, path: string): unknown;
/** Does the checked value satisfy the condition? `previous` is the value from the last run. */
function evaluate(cond: Condition, value: unknown, previous?: unknown): boolean;
/** Replace `{value}` in notification text with the checked value (numbers rounded to 2 decimals). */
function fillTemplate(template: string, value: unknown): string;
```

`changed` is true only when there was a previous value and it differs (JSON comparison). Numeric
ops coerce numeric strings; anything non-numeric compares false.
